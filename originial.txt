    <script>
        const promptInput = document.getElementById('prompt-input');
        const submitBtn = document.getElementById('submit-btn');
        const outputDiv = document.getElementById('output');
        const btnText = submitBtn.querySelector('.btn-text');
        const spinner = submitBtn.querySelector('.spinner');
        const modelDropdown = document.getElementById('model-dropdown');
        const outputTitle = document.getElementById('output-title');

        // Initialize Mermaid for diagram rendering
        mermaid.initialize({
            startOnLoad: true,
            theme: 'default',
            securityLevel: 'loose',
            fontFamily: 'Inter, sans-serif'
        });

        // Model configurations
        const modelConfigs = {
            'claude-3-5-sonnet': { name: 'Claude 3.5 Sonnet', icon: 'fas fa-brain' },
            'gpt-4o': { name: 'GPT-4o', icon: 'fab fa-openai' },
            'gpt-4o-mini': { name: 'GPT-4o Mini', icon: 'fas fa-rocket' },
            'meta-llama/Meta-Llama-3.1-70B-Instruct-Turbo': { name: 'Llama 3.1 70B', icon: 'fas fa-dragon' }
        };

        // Update output title based on selected model
        function updateOutputTitle() {
            const selectedModel = modelDropdown.value;
            const config = modelConfigs[selectedModel];
            if (config) {
                outputTitle.innerHTML = `<i class="${config.icon}"></i> ${config.name}`;
            }
        }

        // Function to set prompt from example buttons
        function setPrompt(prompt) {
            promptInput.value = prompt;
            promptInput.focus();
        }

        // Function to show loading state
        function showLoading() {
            btnText.style.display = 'none';
            spinner.style.display = 'block';
            submitBtn.disabled = true;

            const selectedModel = modelDropdown.value;
            const modelName = modelConfigs[selectedModel]?.name || 'AI';

            // Show typing animation
            outputDiv.innerHTML = `
                <div class="loading-dots">
                    <div class="loading-dot"></div>
                    <div class="loading-dot"></div>
                    <div class="loading-dot"></div>
                </div>
                <span style="margin-left: 10px; color: #6b7280;">${modelName} is thinking...</span>
            `;
        }

        // Function to hide loading state
        function hideLoading() {
            btnText.style.display = 'flex';
            spinner.style.display = 'none';
            submitBtn.disabled = false;
        }

        // Enhanced function to format markdown-like text with better diagram support
        function formatResponse(text) {
            // Handle Mermaid diagrams first
            text = text.replace(/```mermaid\n([\s\S]*?)```/g, (match, diagram) => {
                const diagramId = 'mermaid-' + Math.random().toString(36).substr(2, 9);
                setTimeout(() => {
                    try {
                        mermaid.render(diagramId, diagram.trim(), (svgCode) => {
                            const element = document.getElementById(diagramId);
                            if (element) {
                                element.innerHTML = svgCode;
                            }
                        });
                    } catch (error) {
                        console.error('Mermaid rendering error:', error);
                    }
                }, 100);
                return `<div class="diagram-container"><div id="${diagramId}">Loading diagram...</div></div>`;
            });

            // Handle ASCII art and diagrams
            text = text.replace(/```(ascii|diagram|art)\n([\s\S]*?)```/g, (match, type, content) => {
                return `<div class="ascii-art">${content.trim()}</div>`;
            });

            // Convert markdown to HTML
            let formatted = text
                // Headers
                .replace(/^### (.*$)/gm, '<h3>$1</h3>')
                .replace(/^## (.*$)/gm, '<h2>$1</h2>')
                .replace(/^# (.*$)/gm, '<h1>$1</h1>')

                // Bold text
                .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
                .replace(/__(.*?)__/g, '<strong>$1</strong>')

                // Italic text
                .replace(/\*(.*?)\*/g, '<em>$1</em>')
                .replace(/_(.*?)_/g, '<em>$1</em>')

                // Strikethrough
                .replace(/~~(.*?)~~/g, '<del>$1</del>')

                // Inline code (avoid matching already processed code blocks)
                .replace(/`([^`\n]+)`/g, '<code>$1</code>')

                // Enhanced code blocks with copy functionality and better language detection
                .replace(/```(\w+)?\n([\s\S]*?)```/g, (match, lang, code) => {
                    const language = lang || 'text';
                    const copyId = 'copy-' + Math.random().toString(36).substr(2, 9);
                    return `<pre data-lang="${language}"><button class="copy-btn" onclick="copyCode('${copyId}')">Copy</button><code id="${copyId}">${code.trim()}</code></pre>`;
                })

                // LaTeX formulas (block)
                .replace(/\$\$([\s\S]*?)\$\$/g, '<div class="math-formula">$1</div>')

                // LaTeX formulas (inline)
                .replace(/\$([^$\n]+)\$/g, '<span class="inline-math">$1</span>')

                // Enhanced links
                .replace(/\[([^\]]+)\]\(([^)]+)\)/g, '<a href="$2" target="_blank" rel="noopener noreferrer">$1</a>')

                // Blockquotes with better handling
                .replace(/^> (.*$)/gm, '<blockquote>$1</blockquote>')
                .replace(/(<blockquote>.*<\/blockquote>\n?)+/g, (match) => {
                    return match.replace(/<\/blockquote>\n?<blockquote>/g, '<br>');
                })

                // Horizontal rules
                .replace(/^---+$/gm, '<hr>')
                .replace(/^___+$/gm, '<hr>')
                .replace(/^\*\*\*+$/gm, '<hr>')

                // Task lists
                .replace(/^\s*- \[([ x])\] (.*$)/gm, (match, checked, text) => {
                    const isChecked = checked === 'x' ? 'checked' : '';
                    return `<li class="task-list-item"><span class="task-checkbox ${isChecked}"></span>${text}</li>`;
                })

                // Regular unordered lists
                .replace(/^[\s]*[-*+] (.*$)/gm, '<li>$1</li>')

                // Ordered lists
                .replace(/^\s*\d+\. (.*$)/gm, '<li>$1</li>')

                // Wrap consecutive list items
                .replace(/(<li>.*<\/li>\s*)+/g, (match) => {
                    if (match.includes('task-list-item')) {
                        return `<ul class="task-list">${match}</ul>`;
                    }
                    return `<ul>${match}</ul>`;
                })

                // Tables
                .replace(/^\|(.+)\|\s*\n\|[-\s|:]+\|\s*\n((?:\|.*\|\s*\n?)*)/gm, (match, header, rows) => {
                    const headerCells = header.split('|').map(cell => `<th>${cell.trim()}</th>`).join('');
                    const bodyRows = rows.trim().split('\n').map(row => {
                        const cells = row.split('|').slice(1, -1).map(cell => `<td>${cell.trim()}</td>`).join('');
                        return `<tr>${cells}</tr>`;
                    }).join('');
                    return `<table><thead><tr>${headerCells}</tr></thead><tbody>${bodyRows}</tbody></table>`;
                })

                // Alerts/callouts
                .replace(/^!!! (info|warning|error|success) (.*$)/gm, (match, type, title) => {
                    const icons = {
                        info: 'fas fa-info-circle',
                        warning: 'fas fa-exclamation-triangle',
                        error: 'fas fa-times-circle',
                        success: 'fas fa-check-circle'
                    };
                    return `<div class="alert alert-${type}"><i class="alert-icon ${icons[type]}"></i><div><strong>${title}</strong></div></div>`;
                });

            // Split into paragraphs and wrap appropriately
            const paragraphs = formatted.split(/\n\s*\n/);
            formatted = paragraphs.map(paragraph => {
                paragraph = paragraph.trim();
                if (paragraph &&
                    !paragraph.startsWith('<h') &&
                    !paragraph.startsWith('<ul') &&
                    !paragraph.startsWith('<ol') &&
                    !paragraph.startsWith('<pre') &&
                    !paragraph.startsWith('<blockquote') &&
                    !paragraph.startsWith('<hr') &&
                    !paragraph.startsWith('<div class="math-formula"') &&
                    !paragraph.startsWith('<div class="diagram-container"') &&
                    !paragraph.startsWith('<div class="ascii-art"') &&
                    !paragraph.startsWith('<table') &&
                    !paragraph.startsWith('<div class="alert"')) {
                    return `<p>${paragraph}</p>`;
                }
                return paragraph;
            }).join('\n\n');

            return formatted;
        }

        // Function to copy code to clipboard
        function copyCode(elementId) {
            const codeElement = document.getElementById(elementId);
            if (codeElement) {
                navigator.clipboard.writeText(codeElement.textContent).then(() => {
                    // Show temporary feedback
                    const copyBtn = codeElement.parentElement.querySelector('.copy-btn');
                    const originalText = copyBtn.textContent;
                    copyBtn.textContent = 'Copied!';
                    setTimeout(() => {
                        copyBtn.textContent = originalText;
                    }, 2000);
                });
            }
        }

        // Enhanced function to display formatted text
        function typeFormattedText(text, element) {
            const formattedText = formatResponse(text);
            element.innerHTML = '';

            // Show formatted text with fade-in animation
            element.innerHTML = formattedText;

            // Render any mermaid diagrams
            setTimeout(() => {
                mermaid.run();
            }, 200);

            // Add fade-in animation
            element.style.opacity = '0';
            element.style.transform = 'translateY(10px)';

            setTimeout(() => {
                element.style.transition = 'all 0.5s ease';
                element.style.opacity = '1';
                element.style.transform = 'translateY(0)';
            }, 100);
        }

        // Submit function with model selection and robust error handling
        submitBtn.addEventListener('click', async () => {
            const prompt = promptInput.value.trim();
            if (!prompt) {
                outputDiv.innerHTML = '<div style="color: #ef4444; font-style: italic;">⚠️ Please enter a prompt!</div>';
                return;
            }

            const selectedModel = modelDropdown.value;
            showLoading();

            try {
                // Check if we should use streaming (for Llama model)
                if (selectedModel.includes('llama')) {
                    await handleStreamingResponse(prompt, selectedModel);
                } else {
                    // Regular non-streaming response
                    const response = await puter.ai.chat(prompt, { model: selectedModel });
                    hideLoading();

                    // Handle different response formats
                    if (response && response.message && response.message.content && response.message.content[0]) {
                        typeFormattedText(response.message.content[0].text, outputDiv);
                    } else if (typeof response === 'string') {
                        typeFormattedText(response, outputDiv);
                    } else if (response && response.text) {
                        typeFormattedText(response.text, outputDiv);
                    } else {
                        // Fallback for any other response structure
                        const responseText = JSON.stringify(response, null, 2);
                        typeFormattedText(responseText, outputDiv);
                    }
                }

                // Clear the input after successful response
                promptInput.value = '';
            } catch (error) {
                hideLoading();
                console.error('API Error:', error);

                let errorMessage = `<h3>Error with ${modelConfigs[selectedModel]?.name || selectedModel}</h3>`;
                errorMessage += `<p>${error.message || 'Unknown error occurred'}</p>`;
                errorMessage += '<p>Try selecting a different model from the dropdown.</p>';

                outputDiv.innerHTML = `<div style="color: #ef4444;">${errorMessage}</div>`;
            }
        });

        // Handle streaming responses (primarily for Llama)
        async function handleStreamingResponse(prompt, model) {
            try {
                const streamResponse = await puter.ai.chat(prompt, {
                    model: model,
                    stream: true
                });

                // Clear output area first
                outputDiv.innerHTML = '';

                // Process each chunk as it arrives
                let fullResponse = '';
                for await (const part of streamResponse) {
                    if (part && part.text) {
                        fullResponse += part.text;
                        // Update the display with the current accumulated text
                        typeFormattedText(fullResponse, outputDiv);
                    }
                }

                hideLoading();
            } catch (error) {
                hideLoading();
                throw error; // Let the main error handler deal with it
            }
        }

        // Function to analyze images (for vision capabilities)
        async function analyzeImage(imageUrl, question) {
            try {
                const response = await puter.ai.chat(question, imageUrl);
                return response;
            } catch (error) {
                console.error('Image analysis error:', error);
                throw error;
            }
        }

        // Event listeners
        promptInput.addEventListener('keypress', (e) => {
            if (e.key === 'Enter' && !submitBtn.disabled) {
                submitBtn.click();
            }
        });

        promptInput.addEventListener('input', () => {
            if (promptInput.value.trim()) {
                submitBtn.style.opacity = '1';
                submitBtn.style.transform = 'scale(1.02)';
            } else {
                submitBtn.style.opacity = '0.8';
                submitBtn.style.transform = 'scale(1)';
            }
        });

        modelDropdown.addEventListener('change', updateOutputTitle);

        // Initialize
        updateOutputTitle();
        window.setPrompt = setPrompt;
        window.copyCode = copyCode;

        // Resource Panel Toggle Function
        function toggleResourcePanel() {
            const panel = document.getElementById('resourcePanel');
            panel.classList.toggle('open');
        }

        // Make toggle function globally accessible
        window.toggleResourcePanel = toggleResourcePanel;
    </script>
